* Early init (loaded before init.el)
Disable package.el at startup so straight.el is the only package manager; avoids the "straight.el was loaded when package.el was already loaded" warning.
#+begin_src emacs-lisp :tangle .emacs.d/early-init.el :mkdirp yes
(setq package-enable-at-startup nil)

;; Suppress native-compiler warnings from third-party packages
(setq native-comp-async-report-warnings-errors 'silent)
#+end_src

* Bootstrap
** Straight.el
Bootstrap straight.el first — it is the sole package manager. Package.el is disabled in =early-init.el=.
#+begin_src emacs-lisp
  (defvar bootstrap-version)
  (let ((bootstrap-file
	 (expand-file-name
	  "straight/repos/straight.el/bootstrap.el"
	  (or (bound-and-true-p straight-base-dir)
	      user-emacs-directory)))
	(bootstrap-version 7))
    (unless (file-exists-p bootstrap-file)
      (with-current-buffer
	  (url-retrieve-synchronously
	   "https://raw.githubusercontent.com/radian-software/straight.el/develop/install.el"
	   'silent 'inhibit-cookies)
	(goto-char (point-max))
	(eval-print-last-sexp)))
    (load bootstrap-file nil 'nomessage))
#+end_src

** Use-package
Install use-package via straight.el and make straight the default backend for all =use-package= declarations.
#+begin_src emacs-lisp
  (straight-use-package 'use-package)
  (setq straight-use-package-by-default t)
  (setq use-package-always-defer t)
  (setq use-package-verbose t)
#+end_src

* Load modules
Add the lisp directory to the load path and load each module.

Each module is loaded inside =condition-case= so a failure in one does not abort
the rest of init. Without this, a single bad form part-way through the list
silently prevents every later module from loading at all — which is exactly what
=treemacs-start-on-boot= in =files.el= used to do to =macos.el= and =extras.el=.
Failures are collected and reported once at the end of startup instead of
disappearing.

#+begin_src emacs-lisp
  (add-to-list 'load-path (expand-file-name "lisp" user-emacs-directory))

  (defvar skls/module-load-errors nil
    "Alist of (MODULE . ERROR) for config modules that failed to load.")

  (defun skls/load-module (module)
    "Load MODULE from lisp/, recording any error instead of propagating it."
    (condition-case err
        (load module)
      (error
       (push (cons module err) skls/module-load-errors)
       (message "[init] module `%s' failed: %s" module (error-message-string err)))))

  (dolist (module '("core" "programming" "org-config" "files" "macos" "extras"))
    (skls/load-module module))

  (when skls/module-load-errors
    (add-hook 'emacs-startup-hook
              (lambda ()
                (message "[init] %d module(s) failed to load: %s — see *Messages*"
                         (length skls/module-load-errors)
                         (mapconcat #'car (reverse skls/module-load-errors) ", ")))
              t))
#+end_src

* Auto-tangle on save

Automatically tangle Org config files when saved. Uses =org-auto-tangle= — add =#+auto_tangle: t= to any Org file that should auto-tangle.
#+begin_src emacs-lisp
  (use-package org-auto-tangle
    :defer t
    :hook (org-mode . org-auto-tangle-mode))
#+end_src

* Cleanup
Dial down GC thershold so it runs frequently but in less time.
#+begin_src emacs-lisp
  ;; Make gc pauses faster by decreasing the threshold.
  (setq gc-cons-threshold (* 5 1024 1024))
  (setq read-process-output-max (* 1024 1024)) ;; 1mb

  ;; Start the Emacs server so emacsclient can connect (e.g. git editor).
  (require 'server)
  (unless (server-running-p)
    (server-start))
#+end_src
