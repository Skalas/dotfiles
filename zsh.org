* Aliases
#+begin_src conf :tangle src/skls/aliases
  # Git clone from GitHub
  function gch() {
    git clone git://github.com/skalas/$1.git
  }

  alias grm="git status | grep deleted | awk '{print \$3}' | xargs git rm"

  # Quickly clobber a file and checkout
  function grf() {
    rm $1
    git checkout $1
  }

  # General
  alias h='history'
  alias md='mkdir -p'

  # Crear directorio y un archivo
  mkpfile() { mkdir -p "$(dirname "$1")" || return; touch "$1"; }

  # Package manager aliases: Linux (apt) vs macOS (Homebrew; do not override system install)
  if [[ "$(uname -s)" == Darwin ]]; then
    alias update="brew update && brew upgrade"
  else
    alias install="sudo apt install"
    alias update="sudo apt update && sudo apt upgrade -y && sudo apt autoremove -y"
  fi

  SHELL=/bin/zsh

  # Install Ruby Gems to ~/gems
  export GEM_HOME="$HOME/gems"
  export PATH="$HOME/gems/bin:$PATH"

#+end_src
